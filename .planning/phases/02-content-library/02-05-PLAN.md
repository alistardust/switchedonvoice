  Imports: json, pytest, pathlib.Path, from switchedonvoice.storage.db import init_db; from switchedonvoice.storage.books import BookRepository; from switchedonvoice.services.book_watcher import BookWatcherService

  Module-level constant: MINIMAL_SOV_BOOK (same dict as in test_books_repository.py — define as a module-level fixture or constant to avoid repetition).

  Fixtures:
    @pytest.fixture() def db(tmp_path): p = tmp_path / "test.db"; init_db(p); return p
    @pytest.fixture() def books_dir(tmp_path): d = tmp_path / "books"; return d  # NOT pre-created; BookWatcherService must create it

  Tests (all use qtbot from pytest-qt):
  
  - test_watcher_creates_books_dir(qtbot, db, books_dir):
      assert not books_dir.exists()
      svc = BookWatcherService(books_dir=books_dir, db_path=db)
      qtbot.addWidget(svc)  # register for cleanup
      assert books_dir.exists()
      
  - test_watcher_ingests_valid_drop(qtbot, db, books_dir, tmp_path):
      svc = BookWatcherService(books_dir=books_dir, db_path=db)
      qtbot.addWidget(svc)
      (books_dir / "test.sov-book").write_text(json.dumps(MINIMAL_SOV_BOOK), encoding="utf-8")
      with qtbot.waitSignal(svc.book_ingested, timeout=2000) as blocker:
          pass
      assert blocker.args == ["test-book"]
      repo = BookRepository(db)
      assert len(repo.list_books()) == 1

  - test_watcher_malformed_json_emits_failed_not_ingested(qtbot, db, books_dir):
      svc = BookWatcherService(books_dir=books_dir, db_path=db)
      qtbot.addWidget(svc)
      (books_dir / "bad.sov-book").write_text("not valid json {{{", encoding="utf-8")
      with qtbot.waitSignal(svc.ingest_failed, timeout=2000):
          pass
      repo = BookRepository(db)
      assert len(repo.list_books()) == 0  # nothing ingested

  - test_watcher_skips_duplicate_slug(qtbot, db, books_dir):
      # Pre-ingest the book
      BookRepository(db).ingest(MINIMAL_SOV_BOOK)
      svc = BookWatcherService(books_dir=books_dir, db_path=db)
      qtbot.addWidget(svc)
      (books_dir / "test.sov-book").write_text(json.dumps(MINIMAL_SOV_BOOK), encoding="utf-8")
      with qtbot.waitSignal(svc.book_ingested, timeout=2000):
          pass  # succeeded signal still emits (slug already existed)
      repo = BookRepository(db)
      assert len(repo.list_books()) == 1  # NOT 2

  - test_watcher_missing_key_emits_failed(qtbot, db, books_dir):
      bad_data = {"title": "No slug field"}  # missing "slug" key
      svc = BookWatcherService(books_dir=books_dir, db_path=db)
      qtbot.addWidget(svc)
      (books_dir / "missing-key.sov-book").write_text(json.dumps(bad_data), encoding="utf-8")
      with qtbot.waitSignal(svc.ingest_failed, timeout=2000):
          pass
      assert len(BookRepository(db).list_books()) == 0

Note on QT_QPA_PLATFORM: QFileSystemWatcher uses inotify — it does NOT require a display server. pytest-qt automatically sets offscreen platform for widget tests. Watcher tests run headless without extra configuration.

Note on qtbot.addWidget(svc): BookWatcherService is a QObject, not a QWidget. Use qtbot.addWidget only for QWidgets. For QObjects, use a try/finally cleanup or leave cleanup to Qt parent ownership (svc has no Qt parent unless passed). For test isolation, the simplest approach is: after the test, call svc.deleteLater() in a finalizer or let garbage collection handle it. If qtbot.addWidget raises TypeError for non-QWidget, remove it and add a manual cleanup.

  Actually: replace qtbot.addWidget(svc) with just creating svc as a local variable. Qt parent ownership is None, but Python GC will call ~QObject when svc goes out of scope. This is fine for tests.
