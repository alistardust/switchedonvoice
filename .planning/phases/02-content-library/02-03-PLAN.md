   python -c "
import json
from pathlib import Path

books = [
    ('alice-wonderland.sov-book', 12),
    ('wizard-of-oz.sov-book', 24),
    ('anne-of-green-gables.sov-book', 38),
]
base = Path('src/switchedonvoice/content/books')
for filename, expected_chapters in books:
    data = json.loads((base / filename).read_text(encoding='utf-8'))
    got = len(data['chapters'])
    total_passages = sum(len(ch['passages']) for ch in data['chapters'])
    assert got == expected_chapters, f'{filename}: expected {expected_chapters} chapters, got {got}'
    assert all(p['difficulty'] in ('easy','medium','hard') for ch in data['chapters'] for p in ch['passages'])
    assert all(p['word_count'] > 0 for ch in data['chapters'] for p in ch['passages'])
    print(f'PASS: {filename} — {got} chapters, {total_passages} passages')
"

CHAPTER HEADING NOTES (from research, critical to get right):
