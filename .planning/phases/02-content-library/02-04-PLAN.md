<!-- Qt threading pattern (research Pattern 3) -->
QFileSystemWatcher.addPath(str) → bool   # returns False silently if path does not exist
directoryChanged signal: fires on main thread (thread that owns the watcher)
QThread + moveToThread: background worker; thread.started connects to worker.run()
Cross-thread signals: Qt auto-queues; thread-safe by design

<!-- Critical pitfalls to avoid (from research) -->
- NEVER call addPath() before mkdir() — watcher returns False silently for non-existent dirs
- NEVER create BookRepository with a shared connection — open per-call inside QThread.run()
- NEVER do DB writes in _on_directory_changed() — that is on the main thread
- ALWAYS guard with book_exists(slug) to prevent duplicate ingestion on repeated signals
- ALWAYS connect thread.finished → thread.deleteLater and worker → worker.deleteLater to prevent Qt object leaks
</interfaces>

_IngestionWorker(QObject):
  Signals:
    succeeded = Signal(str)       # emits slug on success
    failed = Signal(str, str)     # emits (path_str, error_message) on any exception
  
  __init__(self, sov_book_path: Path, db_path: Path, parent: QObject | None = None):
    super().__init__(parent)
    self._path = sov_book_path
    self._db_path = db_path
  
  @Slot()
  def run(self) -> None:
    Wrapped in try/except Exception:
      1. Create BookRepository(self._db_path) inside run() — not before (thread-safety: no shared connections).
      2. Read and parse self._path.read_text(encoding="utf-8") using json.loads().
      3. slug = data["slug"]
      4. If repo.book_exists(slug): log info "book_already_ingested", emit succeeded(slug), return.
      5. repo.ingest(data)
      6. _log.info("book_ingested", slug=slug, path=str(self._path))
      7. emit succeeded(slug)
    Except Exception as exc:
      _log.error("book_ingest_failed", path=str(self._path), error=str(exc))
      emit failed(str(self._path), str(exc))

BookWatcherService(QObject):
  Signals:
    book_ingested = Signal(str)   # emits slug — UI notification hook for Phase 3
    ingest_failed = Signal(str)   # emits error_message
  
  __init__(self, books_dir: Path, db_path: Path, parent: QObject | None = None):
    super().__init__(parent)
    self._books_dir = books_dir
    self._db_path = db_path
    self._watcher = QFileSystemWatcher(self)
    
    books_dir.mkdir(parents=True, exist_ok=True)  # MUST precede addPath (Pitfall 3)
    result = self._watcher.addPath(str(books_dir))
    if not result:
        _log.warning("watcher_add_path_failed", path=str(books_dir))
    self._watcher.directoryChanged.connect(self._on_directory_changed)
  
  @Slot(str)
  def _on_directory_changed(self, _path: str) -> None:
    Iterate self._books_dir.glob("*.sov-book") and call self._ingest_async(f) for each.
    Note: book_exists() guard is inside _IngestionWorker.run(), not here — keeping _on_directory_changed fast.
  
  def _ingest_async(self, path: Path) -> None:
    worker = _IngestionWorker(path, self._db_path)
    thread = QThread(self)
    worker.moveToThread(thread)
    thread.started.connect(worker.run)
    worker.succeeded.connect(self.book_ingested)
    worker.failed.connect(lambda p, e: self.ingest_failed.emit(e))
    worker.succeeded.connect(thread.quit)
    worker.failed.connect(thread.quit)
    worker.succeeded.connect(worker.deleteLater)
    worker.failed.connect(worker.deleteLater)
    thread.finished.connect(thread.deleteLater)
    thread.start()
